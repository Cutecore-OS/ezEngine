ez_cmake_init()
option(EZ_BUILD_MOVIES_PLUGIN "Build the Render Movie plugin (requires MiniAudio and FFmpeg)" OFF)
if(NOT EZ_BUILD_MOVIES_PLUGIN)
  return()
endif()
ez_requires_miniaudio()
set(EZ_MOVIES_FFMPEG_ROOT "" CACHE PATH "FFmpeg SDK with include, lib and bin directories")
find_path(MOVIES_FFMPEG_INCLUDE_DIR libavcodec/avcodec.h HINTS "${EZ_MOVIES_FFMPEG_ROOT}/include" REQUIRED)
get_filename_component(PROJECT_NAME ${CMAKE_CURRENT_SOURCE_DIR} NAME_WE)
ez_create_target(LIBRARY ${PROJECT_NAME})
target_link_libraries(${PROJECT_NAME} PUBLIC GameEngine PRIVATE MiniAudioPlugin MiniAudio)
target_include_directories(${PROJECT_NAME} PRIVATE "${MOVIES_FFMPEG_INCLUDE_DIR}")
foreach(LIB avcodec avformat avutil swscale)
  find_library(MOVIES_${LIB}_LIBRARY NAMES ${LIB} HINTS "${EZ_MOVIES_FFMPEG_ROOT}/lib" REQUIRED)
  target_link_libraries(${PROJECT_NAME} PRIVATE "${MOVIES_${LIB}_LIBRARY}")
endforeach()
if(WIN32)
  file(GLOB MOVIES_FFMPEG_DLLS "${EZ_MOVIES_FFMPEG_ROOT}/bin/*.dll")
  foreach(DLL ${MOVIES_FFMPEG_DLLS})
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" "$<TARGET_FILE_DIR:${PROJECT_NAME}>")
  endforeach()
endif()
ez_copy_plugin_bundle(${PROJECT_NAME} "Movies")
